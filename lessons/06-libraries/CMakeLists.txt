cmake_minimum_required(VERSION 3.28)
project(AssetLens VERSION 1.0.0 LANGUAGES CXX)
add_library(assetlens_headers INTERFACE)
target_include_directories(assetlens_headers INTERFACE include)
target_compile_features(assetlens_headers INTERFACE cxx_std_17)
add_library(assetlens_objects OBJECT src/scanner.cpp src/report.cpp)
target_link_libraries(assetlens_objects PUBLIC assetlens_headers)
set_target_properties(assetlens_objects PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(assetlens_static STATIC $<TARGET_OBJECTS:assetlens_objects>)
add_library(assetlens_shared SHARED $<TARGET_OBJECTS:assetlens_objects>)
set_target_properties(assetlens_shared PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_link_libraries(assetlens_static PUBLIC assetlens_headers)
target_link_libraries(assetlens_shared PUBLIC assetlens_headers)
add_executable(assetlens app/main.cpp)
target_link_libraries(assetlens PRIVATE assetlens_static)
add_executable(assetlens_dynamic app/main.cpp)
target_link_libraries(assetlens_dynamic PRIVATE assetlens_shared)
