cmake_minimum_required(VERSION 3.28)
project(AssetLens VERSION 1.0.0 LANGUAGES CXX)
add_executable(assetlens app/main.cpp)
target_sources(assetlens PRIVATE src/scanner.cpp src/report.cpp)
target_include_directories(assetlens PRIVATE include)
target_compile_features(assetlens PRIVATE cxx_std_17)
target_compile_definitions(assetlens PRIVATE ASSETLENS_APPLICATION=1)
if(MSVC)
  target_compile_options(assetlens PRIVATE /W4 /utf-8)
else()
  target_compile_options(assetlens PRIVATE -Wall -Wextra -Wpedantic)
endif()
