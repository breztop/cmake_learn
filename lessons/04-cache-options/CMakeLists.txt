cmake_minimum_required(VERSION 3.28)
project(AssetLens VERSION 1.0.0 LANGUAGES CXX)
option(ASSETLENS_ENABLE_JSON "Enable JSON report output" ON)
configure_file(cmake/config.hpp.in generated/assetlens/config.hpp @ONLY)
add_executable(assetlens app/main.cpp src/scanner.cpp src/report.cpp)
target_include_directories(assetlens PRIVATE include)
target_compile_features(assetlens PRIVATE cxx_std_17)
target_include_directories(assetlens PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(assetlens PRIVATE ASSETLENS_HAS_CONFIG)
