cmake_minimum_required(VERSION 3.28)
project(AssetLens VERSION 1.0.0 LANGUAGES CXX)
set(asset_sources src/scanner.cpp)
list(APPEND asset_sources src/report.cpp)
foreach(source IN LISTS asset_sources)
  message(STATUS "Source: ${source}")
endforeach()
list(LENGTH asset_sources source_count)
message(STATUS "Library sources: ${source_count}")
include(cmake/Paths.cmake)
add_executable(assetlens app/main.cpp ${asset_sources})
target_include_directories(assetlens PRIVATE include)
target_compile_features(assetlens PRIVATE cxx_std_17)
message(STATUS "Project source: ${PROJECT_SOURCE_DIR}")
message(STATUS "Project build: ${PROJECT_BINARY_DIR}")
message(STATUS "Current source: ${CMAKE_CURRENT_SOURCE_DIR}")
